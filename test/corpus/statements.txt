function control(limit) {
  var x = 0, y = 1;
  for (var i = 0, j = limit; i < j; i++, --j) {
    if (i == 2) { continue; } else if (j == 3) { break; } else { x += i; }
  }
  for (x = 0, y = limit; x < y; x += 1, y -= 1) { x *= 2; }
  for (;;) { break; }
  while (x < limit) { ++x; }
  do { x--; } while (x > 0);
  { var nested = x; }
  return x;
}

    case instanceof Number:
    case 0: { return true; }
    case :ready: value = null; break;
    default: return false;
  }

  catch (ex instanceof Lang.Exception) { throw ex; }
  catch (other) { return null; }
  finally { cleanup(); }
  try { run(); } finally { cleanup(); }
